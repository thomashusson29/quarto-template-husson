\NewMarkClass{hussonnav}

% State definitions: \hussonstate<id> -> {H1_id}{Bar content}
\newcommand{\defstate}[3]{%
  \expandafter\def\csname hussonh@#1\endcsname{#2}%
  \expandafter\def\csname hussonbar@#1\endcsname{#3}%
}

\defstate{1}{1}{CHAPITRE 1 -- Sec A}
\defstate{2}{1}{CHAPITRE 1 -- Sec B}
\defstate{3}{2}{CHAPITRE 2 -- Sec C}
\defstate{4}{2}{CHAPITRE 2 -- Sec D}

\makeatletter
\newcommand{\getstateHone}[1]{%
  \@ifundefined{hussonh@#1}{0}{\csname hussonh@#1\endcsname}%
}
\newcommand{\getstateBar}[1]{%
  \@ifundefined{hussonbar@#1}{}{\csname hussonbar@#1\endcsname}%
}

\newcommand{\resolveactivemark}{%
  \edef\@topm{\TopMark{hussonnav}}%
  \edef\@firstm{\FirstMark{hussonnav}}%
  \ifx\@topm\@empty
    \@firstm
  \else
    \edef\@toph{\getstateHone{\@topm}}%
    \edef\@firsth{\getstateHone{\@firstm}}%
    \ifx\@toph\@firsth
      \@topm
    \else
      \@firstm
    \fi
  \fi
}

\newcommand{\renderhussonbar}{%
  \edef\@curstate{\resolveactivemark}%
  \getstateBar{\@curstate}%
}
\makeatother
